let state = {

  currentMode: '',

  currentIdx: 0,

  todayWords: [],      // 今天要学的5个词

// 从localStorage加载

function loadState() {

  const saved = localStorage.getItem('englishApp_state');

  if (saved) {

    state = { ...state, ...JSON.parse(saved) };

  }

  updateHomeUI();

}

function saveState() {

  localStorage.setItem('englishApp_state', JSON.stringify(state));

}


// ============ 语音合成 ============

function speak(text) {

  if ('speechSynthesis' in window) {

    // 停止当前正在播放的语音

    speechSynthesis.cancel();

    const utter = new SpeechSynthesisUtterance(text);

    utter.lang = 'en-US';

    utter.rate = 0.7;  // 慢一点，适合孩子

    utter.pitch = 1.2; // 高一点，更可爱

    speechSynthesis.speak(utter);

  } else {

    alert('你的浏览器不支持语音功能，试试Chrome或Safari吧！');

  }

}

function speakCurrent() {

  const word = state.currentList[state.currentIdx];

  if (word) speak(word.en);

}


// ============ 首页 ============

  document.getElementById('totalStars').textContent = state.stars;

  document.getElementById('dayNum').textContent = state.day;

  document.getElementById('masteredCount').textContent = state.masteredWords.length;

  const totalWords = wordData.length;

  const pct = totalWords > 0 ? (state.masteredWords.length / totalWords * 100) : 0;

  document.getElementById('progressFill').style.width = pct + '%';

}


function goHome() {

  document.getElementById('homePage').classList.remove('hidden');

  document.getElementById('learnPage').classList.add('hidden');

  document.getElementById('listenPage').classList.add('hidden');

  document.getElementById('resultPage').classList.add('hidden');

  document.getElementById('listPage').classList.add('hidden');

}


// ============ 学习模式 ============

function startMode(mode) {

  state.currentMode = mode;

  document.getElementById('homePage').classList.add('hidden');


  if (mode === 'learn') {

    // 获取今天的5个词（按day字段）

    const todayData = wordData.filter(w => w.day === state.day);

    if (todayData.length === 0) {

      // 如果今天的学完了，从错题里取

      if (state.wrongWords.length > 0) {

        state.currentList = [...state.wrongWords];

      } else {

        // 全部学完了

        showResult('🎊 本周全部学完啦！', '你已经掌握了所有单词！', '⭐⭐⭐⭐⭐');

        return;

      }

    } else {

      state.currentList = todayData;

    }

    state.currentIdx = 0;

    document.getElementById('learnPage').classList.remove('hidden');

    document.getElementById('modeTitle').textContent = '📖 学新单词';

    showCurrentWord();


  } else if (mode === 'review') {

    if (state.wrongWords.length === 0) {

      alert('🎉 没有错题！你太厉害了！');

      goHome();

      return;

    }

    state.currentList = [...state.wrongWords];

    state.currentIdx = 0;

    document.getElementById('learnPage').classList.remove('hidden');

    document.getElementById('modeTitle').textContent = '🔄 复习错题';

    showCurrentWord();


  } else if (mode === 'listen') {

    if (state.masteredWords.length + state.wrongWords.length === 0) {

      alert('先学几个单词再来玩听音游戏吧！');

      goHome();

      return;

    }

    document.getElementById('listenPage').classList.remove('hidden');

    startListenRound();

  }

}


function showCurrentWord() {

  const word = state.currentList[state.currentIdx];

  if (!word) {

    // 学完了

    showSessionResult();

    return;

  }


  document.getElementById('wordEnglish').textContent = word.en;

  document.getElementById('wordChinese').textContent = `📝 ${word.zh}`;

  document.getElementById('wordSentence').textContent = word.sentence;

  document.getElementById('currentIndex').textContent = state.currentIdx + 1;

  document.getElementById('totalIndex').textContent = state.currentList.length;


  const pct = ((state.currentIdx + 1) / state.currentList.length) * 100;

  document.getElementById('learnProgress').style.width = pct + '%';


  // 掌握状态标记

  const badge = document.getElementById('masteryBadge');

  if (state.masteredWords.includes(word.en)) {

    badge.innerHTML = '<span class="mastery-badge mastered">✅ 已掌握</span>';

  } else if (state.wrongWords.includes(word.en)) {

    badge.innerHTML = '<span class="mastery-badge learning">🔄 复习中</span>';

  } else {

    badge.innerHTML = '';

  }


  // 自动朗读

  setTimeout(() => speak(word.en), 300);

}


function markCorrect() {

  const word = state.currentList[state.currentIdx];

  // 从错题中移除，加入已掌握

  state.wrongWords = state.wrongWords.filter(w => w !== word.en);

  if (!state.masteredWords.includes(word.en)) {

    state.masteredWords.push(word.en);

    state.stars += 2;

  }

  saveState();

  nextWord();

}


function markWrong() {

  const word = state.currentList[state.currentIdx];

  if (!state.wrongWords.includes(word.en)) {

    state.wrongWords.push(word.en);

  }

  saveState();

  // 播放正确发音让孩子再听一次

  speak(word.en);

  setTimeout(() => nextWord(), 800);

}


function nextWord() {

  state.currentIdx++;

  if (state.currentIdx >= state.currentList.length) {

    showSessionResult();

  } else {

    showCurrentWord();

  }

}


function showSessionResult() {

  const correct = state.currentList.filter(w => state.masteredWords.includes(w.en)).length;

  const total = state.currentList.length;

  const stars = '⭐'.repeat(Math.min(5, Math.ceil(correct / total * 5)));

  showResult(

    correct === total ? '🎉 全对！太棒了！' : `💪 不错哦！${correct}/${total} 个对了`,

    `今天学了 ${total} 个词，还有 ${state.wrongWords.length} 个需要复习`,

    stars

  );

}


function showResult(title, text, stars) {

  document.getElementById('learnPage').classList.add('hidden');

  document.getElementById('listenPage').classList.add('hidden');

  document.getElementById('resultPage').classList.remove('hidden');

  document.getElementById('resultTitle').textContent = title;

  document.getElementById('resultText').textContent = text;

  document.getElementById('resultScore').textContent = stars;

}


// ============ 听音识词 ============

function startListenRound() {

  // 从已学和正在学的词里随机选一个

  const pool = wordData.filter(w => state.masteredWords.includes(w.en) || state.wrongWords.includes(w.en) || w.day <= state.day);

  if (pool.length < 2) {

    document.getElementById('listenCard').innerHTML = '<h3>先多学几个单词再来玩吧！</h3><button class="btn btn-correct" onclick="goHome()">返回</button>';

    return;

  }


  const answer = pool[Math.floor(Math.random() * pool.length)];

  state.listenAnswer = answer.en;


  // 生成选项（1个正确 + 3个错误）

  const options = [answer];

  while (options.length < 4) {

    const candidate = pool[Math.floor(Math.random() * pool.length)];

    if (!options.find(o => o.en === candidate.en)) {

      options.push(candidate);

    }

  }

  // 打乱

  options.sort(() => Math.random() - 0.5);


  document.getElementById('listenCard').innerHTML = `

    <button class="btn btn-speak" onclick="playListenAudio()" style="font-size:24px; padding:20px;">🔊 点击听发音</button>

    <div style="display:flex; flex-direction:column; gap:10px; margin-top:15px;">

      ${options.map(o => `<button class="mode-btn" onclick="checkListenAnswer('${o.en}', '${answer.en}')" style="font-size:16px;">${o.zh}</button>`).join('')}

    </div>

  `;


  setTimeout(() => speak(answer.en), 300);

}


function playListenAudio() {

  if (state.listenAnswer) {

    const word = wordData.find(w => w.en === state.listenAnswer);

    if (word) speak(word.en);

  }

}


function checkListenAnswer(selected, correct) {

  if (selected === correct) {

    state.stars += 1;

    saveState();

    // 答对动画

    event.target.style.background = '#56ab2f';

    event.target.style.color = 'white';

    setTimeout(() => {

      // 继续下一题

      startListenRound();

    }, 800);

  } else {

    event.target.style.background = '#ff6b6b';

    event.target.style.color = 'white';

    // 标记到错题本

    if (!state.wrongWords.includes(correct)) {

      state.wrongWords.push(correct);

    }

    saveState();

    setTimeout(() => {

      // 显示正确答案

      const word = wordData.find(w => w.en === correct);

      alert(`正确答案是：${word.zh} (${word.en})`);

      startListenRound();

    }, 800);

  }

}


// ============ 录音功能 ============

function toggleRecord() {

  const btn = document.getElementById('recordBtn');

  if (!state.recording) {

    // 开始录音

    if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {

      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

      const recognition = new SpeechRecognition();

      recognition.lang = 'en-US';

      recognition.continuous = false;

      recognition.interimResults = false;


      btn.classList.add('recording');

      btn.textContent = '⏹';

      state.recording = true;


      recognition.start();


      recognition.onresult = function(event) {

        const transcript = event.results[0][0].transcript.toLowerCase().trim();

        const currentWord = state.currentList[state.currentIdx].en.toLowerCase();

        btn.classList.remove('recording');

        btn.textContent = '🎤';

        state.recording = false;


        if (transcript.includes(currentWord) || currentWord.includes(transcript)) {

          alert('🎉 读得真好！');

          markCorrect();

        } else {

          alert(`你读的是："${transcript}"\n再来一次试试！`);

        }

      };


      recognition.onerror = function() {

        btn.classList.remove('recording');

        btn.textContent = '🎤';

        state.recording = false;

        alert('没听清，再试一次吧！');

      };


      recognition.onend = function() {

        btn.classList.remove('recording');

        btn.textContent = '🎤';

        state.recording = false;

      };

    } else {

      alert('你的浏览器不支持语音识别，但你可以跟着读，然后点"我会了"！');

    }

  }

}

function showWordList() {

  document.getElementById('homePage').classList.add('hidden');

  document.getElementById('listPage').classList.remove('hidden');


  const content = document.getElementById('wordListContent');

  // 按天分组

  let html = '';

  for (let d = 1; d <= 7; d++) {

    const dayWords = wordData.filter(w => w.day === d);

    html += `<div class="card" style="margin:10px 0;">

      <h3 style="color:#667eea; margin-bottom:8px;">第${d}天 - ${dayWords[0]?.category || ''}</h3>`;

    dayWords.forEach(w => {

      const status = state.masteredWords.includes(w.en) ? '✅' : state.wrongWords.includes(w.en) ? '🔄' : '⬜';

      html += `<div style="display:flex; justify-content:space-between; padding:6px; border-bottom:1px solid #eee; font-size:14px;">

        <span>${status} <b>${w.en}</b></span>

        <span style="color:#888;">${w.zh}</span>

      </div>`;

    });

    html += `</div>`;

  }

  content.innerHTML = html;

}
